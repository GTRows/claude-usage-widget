const ANTHROPIC_MESSAGES_ENDPOINT = 'https://api.anthropic.com/v1/messages';

  model: AUTO_FIRE_API_MODEL,
  max_tokens: 1,
  messages: [{ role: 'user', content: '.' }],
};

Module body, in order:
1. Guard: if `!apiKey` return `{ status: 'skipped', reason: 'missing-credentials' }`. Mirrors `web-channel.js` line 49–51 exactly so the dispatcher's success/error counters behave identically.
2. Resolve `fetchImpl`. Default to global `fetch` (Electron 28 ships Node 18+, `fetch` is global). Add a defensive fallback: `const f = fetchImpl || (typeof fetch !== 'undefined' ? fetch : null); if (!f) return { status: 'error', reason: 'transport: no-fetch', latencyMs: 0 };`.
3. `const t0 = now();`
4. POST to `ANTHROPIC_MESSAGES_ENDPOINT` with headers `{ 'content-type': 'application/json', 'x-api-key': apiKey, 'anthropic-version': ANTHROPIC_API_VERSION }` and body `JSON.stringify(AUTO_FIRE_API_PAYLOAD)`. No streaming; explicit `signal` from `AbortSignal.timeout(30000)` if available, else no signal (don't fail the call when the helper is missing).
5. On `res.ok` → `{ status: 'ok', latencyMs: now() - t0 }`.
6. On non-ok response: read `res.status`. If `401` or `403` → `{ status: 'error', reason: 'auth-failed', latencyMs: now() - t0 }` (parallel to `web-channel.js`'s `'session-expired'`). 7. On thrown error (network, abort, etc.): `{ status: 'error', reason: 'transport: ' + (err.message || String(err)), latencyMs: now() - t0 }`. NEVER throw — the dispatcher relies on every channel returning a result object.

Style: 2-space indent, single quotes, semicolons, `'use strict';`, no JSDoc, no `console.log`. Top-of-file comment block (5–10 lines) explaining that this is the API-key channel sibling of `web-channel.js`, why it does not share a generic helper (the two channels have different auth headers and different error-classification rules so factoring is premature), and that the model + payload constants are intentionally pinned.

Module size target: ~80–100 lines, well under the 200-line cap.

Exports: `module.exports = { fireViaApiKey, ANTHROPIC_MESSAGES_ENDPOINT, ANTHROPIC_API_VERSION, AUTO_FIRE_API_MODEL, AUTO_FIRE_API_PAYLOAD };` (export the constants so tests and (eventually) UI hint strings can reference the same source of truth).

Add the following near the existing helper block, right after `getCredentialsForAutoFire()` (currently main.js lines 78–95):

```js
// Anthropic API key storage. Same policy as sessionKey: encrypt with the OS
// keychain via safeStorage when available, fall back to plain electron-store
// only when isEncryptionAvailable() is false (e.g. headless Linux without a
// keyring service). Reads return null when no key is stored.
function readStoredApiKey() {
  if (safeStorage.isEncryptionAvailable()) {
    const encrypted = store.get('apiKey_encrypted');
    if (!encrypted) return null;
    try {
      return safeStorage.decryptString(Buffer.from(encrypted, 'base64'));
    } catch (err) {
      console.error('[Keychain] Failed to decrypt API key:', err.message);
      return null;
    }
  }
  return store.get('apiKey') || null;
}

function writeStoredApiKey(apiKey) {
  if (typeof apiKey !== 'string' || apiKey.length === 0) {
    store.delete('apiKey');
    store.delete('apiKey_encrypted');
    return;
  }
  if (safeStorage.isEncryptionAvailable()) {
    const encrypted = safeStorage.encryptString(apiKey);
    store.set('apiKey_encrypted', encrypted.toString('base64'));
    store.delete('apiKey');
  } else {
    store.set('apiKey', apiKey);
    store.delete('apiKey_encrypted');
  }
}
```

Do NOT inline these into existing handlers. Do NOT add IPC handlers in this plan — plan 02-02 will add `save-api-key` / `clear-api-key` IPC. The functions must be `function` declarations (not consts) so they hoist above the dispatcher construction block at lines 97–105.

async function getCredentialsForAutoFire() {
  let sessionKey = null;
